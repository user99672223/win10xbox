# drivers: the in-process driver host (D16). A user-mode ntoskrnl/hal export
# surface, the DRIVER_OBJECT/DEVICE_OBJECT/IRP engine and object manager, and a
# minimal KMDF shim. Test drivers and tests live in tests/drivers.
#
#   win10xbox_drivers   static library; links win10xbox_core (loader, dispatcher, trace, clock)
#   include/ntk/ntk.h   kernel-mode types for drivers built against the host
#   include/win10xbox/drivers.h   public API (see drivers/README.md)
include(${CMAKE_CURRENT_LIST_DIR}/sources.cmake)

# KMDF function-table indices, parsed from the public KMDF 1.33 header so the shim names its stubs and
# registers implementations by name (no hand-copied numbers). Without the submodule every slot is
# anonymous and the shim implements nothing.
set(WIN10XBOX_WDF_INCLUDE ${CMAKE_SOURCE_DIR}/third_party/Windows-Driver-Frameworks/src/publicinc/wdf/kmdf/1.33)
set(WIN10XBOX_GENERATED_DIR ${CMAKE_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${WIN10XBOX_GENERATED_DIR})
if(EXISTS ${WIN10XBOX_WDF_INCLUDE}/wdffuncenum.h)
  file(STRINGS ${WIN10XBOX_WDF_INCLUDE}/wdffuncenum.h _wdf_lines REGEX "Wdf[A-Za-z0-9_]+TableIndex[ \t]*=[ \t]*[0-9]+")
  set(_wdf_inc "// generated by drivers/CMakeLists.txt from wdffuncenum.h\n")
  foreach(_l IN LISTS _wdf_lines)
    if(_l MATCHES "(Wdf[A-Za-z0-9_]+)TableIndex[ \t]*=[ \t]*([0-9]+)")
      string(APPEND _wdf_inc "WDF_FUNC(${CMAKE_MATCH_1}, ${CMAKE_MATCH_2})\n")
    endif()
  endforeach()
  set(_wdf_old "")
  if(EXISTS ${WIN10XBOX_GENERATED_DIR}/wdf_func_index.inc)
    file(READ ${WIN10XBOX_GENERATED_DIR}/wdf_func_index.inc _wdf_old)
  endif()
  if(NOT "${_wdf_old}" STREQUAL "${_wdf_inc}")
    file(WRITE ${WIN10XBOX_GENERATED_DIR}/wdf_func_index.inc "${_wdf_inc}")
  endif()
  # Linux is case-sensitive but a few KMDF headers include each other with the wrong case.
  if(CMAKE_HOST_UNIX)
    set(WIN10XBOX_KMDF_COMPAT_DIR ${WIN10XBOX_GENERATED_DIR}/kmdf_compat)
    file(MAKE_DIRECTORY ${WIN10XBOX_KMDF_COMPAT_DIR})
    foreach(_pair "WdfQueryInterface.h:wdfqueryinterface.h" "WdfWMI.h:wdfwmi.h" "wdfCommonBuffer.h:wdfcommonbuffer.h"
                  "wdfDmaEnabler.h:wdfdmaenabler.h" "wdfDmaTransaction.h:wdfdmatransaction.h")
      string(REPLACE ":" ";" _p "${_pair}")
      list(GET _p 0 _wrong)
      list(GET _p 1 _right)
      file(WRITE ${WIN10XBOX_KMDF_COMPAT_DIR}/${_wrong} "#include \"${_right}\"\n")
    endforeach()
  endif()
else()
  message(STATUS "drivers: KMDF headers not found (third_party/Windows-Driver-Frameworks), WDF shim has no named functions")
endif()

add_library(win10xbox_drivers STATIC ${WIN10XBOX_DRIVERS_SOURCES})
target_include_directories(win10xbox_drivers
  PUBLIC ${CMAKE_CURRENT_LIST_DIR}/include
  PRIVATE ${CMAKE_CURRENT_LIST_DIR}/ntoskrnl ${WIN10XBOX_GENERATED_DIR})
target_link_libraries(win10xbox_drivers PUBLIC win10xbox_core)

# Import library a driver links against: `import ntoskrnl.exe!Name`.
include(${CMAKE_CURRENT_LIST_DIR}/import_lib.cmake)

  enable_testing()   # the top level enables it after this directory; ctest needs it here to descend
  add_subdirectory(${CMAKE_SOURCE_DIR}/tests/drivers ${CMAKE_BINARY_DIR}/tests/drivers)
endif()
