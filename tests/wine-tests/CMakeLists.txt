# Wine conformance-test subset (M3): a few of Wine's own tests (LGPL), compiled
# as guest PEs with llvm-mingw and run through win10xbox to record a fidelity
# baseline. See SUBSET.md (why these tests) and docs/TESTING.md (the numbers).
#
# The Wine sources are NOT copied into this repository: they are compiled from
# wine-pe/wine-src (wine-pe/fetch-wine.sh) at configure/build time, and the
# resulting guests are only ever built into the build tree (bin/guest-wine/).
# Without wine-src no guest is built and every wine_conformance test SKIPs (exit
# code 77), like the guest e2e tests do without the Wine builtins or NLS data.
# With MSVC, which cannot build them at all, the tests are not registered.
#
# Every test carries the ctest label `wine_conformance` and checks a recorded
# baseline instead of "all green": most of Wine's tests fail or crash on the
# incomplete layer, which is the point. A test fails only on a REGRESSION
# against the baseline in the table below.
set(WIN10XBOX_WINE_SRC_DIR ${CMAKE_SOURCE_DIR}/wine-pe/wine-src CACHE PATH
    "Wine source tree (wine-pe/fetch-wine.sh) providing dlls/*/tests and include/")
# Headers Wine generates from .idl files (widl); only some tests need them.
set(WIN10XBOX_WINE_GEN_INCLUDE_DIR ${CMAKE_SOURCE_DIR}/wine-pe/build/include CACHE PATH
    "Wine's generated headers (wine-pe/build/include after wine-pe/build.sh)")
set(WIN10XBOX_WINE_TEST_TIMEOUT 60 CACHE STRING "Per-test cap in seconds; a longer run is recorded as 'timeout'")

set(_machine ${WIN10XBOX_MACHINE_DIR})
set(_guest_dir ${CMAKE_BINARY_DIR}/bin/guest-wine)
set(_results_dir ${CMAKE_BINARY_DIR}/wine-tests-results)

# ---- can the guests be built here? -------------------------------------------
set(_can_build OFF)
set(_why "")
if(MSVC)
  set(_why "the Wine tests build only with clang/llvm-mingw (Wine's headers + libc stand-in), not MSVC")
elseif(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
  set(_why "needs clang (llvm-mingw); compiler is ${CMAKE_C_COMPILER_ID}")
elseif(NOT EXISTS ${WIN10XBOX_WINE_SRC_DIR}/include/wine/test.h)
  set(_why "${WIN10XBOX_WINE_SRC_DIR} missing (run wine-pe/fetch-wine.sh)")
else()
  execute_process(COMMAND ${CMAKE_C_COMPILER} -print-resource-dir
    OUTPUT_VARIABLE _clang_resource_dir OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _rd_rc)
  if(_rd_rc EQUAL 0 AND IS_DIRECTORY "${_clang_resource_dir}/include")
    set(_can_build ON)
  else()
    set(_why "cannot determine the clang resource directory")
  endif()
endif()
if(MSVC)
  # Not a skip but a toolchain fact: with MSVC these guests can never be built, so no test is registered (a test
  # that can only SKIP would fail CI's ctest gate, tools/ci/check_ctest_run.py). The subset runs, gated, on the
  # llvm-mingw + Wine job (ci.yml linux-cross); see docs/TESTING.md.
  message(STATUS "wine-tests: ${_why}; no wine_conformance tests in this build")
  return()
endif()
if(_can_build)
  message(STATUS "wine-tests: building Wine conformance guests from ${WIN10XBOX_WINE_SRC_DIR}")
else()
  message(STATUS "wine-tests: guests not built (${_why}); wine_conformance tests will SKIP")
endif()
set(_have_gen OFF)
if(EXISTS ${WIN10XBOX_WINE_GEN_INCLUDE_DIR}/unknwn.h)
  set(_have_gen ON)
endif()

if(_can_build)
  add_library(wine_test_support STATIC wine_support.c)   # normal warnings: our code
  set_target_properties(wine_test_support PROPERTIES ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib/guest)
endif()

# win10xbox_wine_test(<dll> <test> <expect> <max_failures> [NEEDS_GEN])
#   expect       run      completes with the "N tests executed" line, failures <= max_failures
#                crash    does not complete (unhandled exception / main thread ended early)
#                timeout  still running after the per-test cap
#   NEEDS_GEN    the source includes headers generated from .idl (needs WIN10XBOX_WINE_GEN_INCLUDE_DIR)
function(win10xbox_wine_test dll test expect max_failures)
  set(name ${dll}_${test})
  set(src ${WIN10XBOX_WINE_SRC_DIR}/dlls/${dll}/tests/${test}.c)
  if(_can_build AND (_have_gen OR NOT "NEEDS_GEN" IN_LIST ARGN) AND EXISTS ${src})
    # Wine's test.h, STANDALONE: one generated main() per test binary.
    set(tl ${CMAKE_CURRENT_BINARY_DIR}/gen/${name}_testlist.c)
    file(GENERATE OUTPUT ${tl} CONTENT
"/* generated by tests/wine-tests/CMakeLists.txt */
#define WIN32_LEAN_AND_MEAN
#include <windows.h>
#define STANDALONE
#include \"wine/test.h\"
extern void func_${test}(void);
const struct test winetest_testlist[] = { { \"${test}\", func_${test} }, { 0, 0 } };
")
    add_executable(guest_wine_${name} ${src} ${tl})
    set_target_properties(guest_wine_${name} PROPERTIES
      OUTPUT_NAME ${name} PREFIX ""
      RUNTIME_OUTPUT_DIRECTORY ${_guest_dir}
      C_STANDARD 17 C_EXTENSIONS ON)
    # Wine's headers replace the toolchain's (as in Wine's own PE build). -w:
    # these are Wine's sources, not ours. The -U drop the repository-wide
    # definitions (UNICODE, _WIN32_WINNT, ...) that would change what the Wine
    # sources mean; options come after definitions on the command line.
    set(_inc ${WIN10XBOX_WINE_SRC_DIR}/include)
    if(_have_gen)
      list(APPEND _inc ${WIN10XBOX_WINE_GEN_INCLUDE_DIR})
    endif()
    list(APPEND _inc ${WIN10XBOX_WINE_SRC_DIR}/include/msvcrt)
    target_include_directories(guest_wine_${name} PRIVATE ${_inc})
    target_compile_definitions(guest_wine_${name} PRIVATE __WINESRC__ _NTSYSTEM_ WINE_NO_LONG_TYPES _UCRT)
    target_compile_options(guest_wine_${name} PRIVATE -w -nostdinc -isystem ${_clang_resource_dir}/include
      -UUNICODE -U_UNICODE -UNOMINMAX -UWIN32_LEAN_AND_MEAN -U_WIN32_WINNT -UWINVER)
    target_link_options(guest_wine_${name} PRIVATE -Wno-unused-command-line-argument)   # repo-wide -static-libstdc++ is moot for C
    target_link_libraries(guest_wine_${name} PRIVATE wine_test_support
      ntdll kernel32 user32 advapi32 gdi32 shell32 ole32 version ws2_32 shlwapi uuid winmm)
    set_property(GLOBAL APPEND PROPERTY WIN10XBOX_WINE_GUESTS guest_wine_${name})
  endif()

  add_test(NAME wine_conformance.${dll}.${test}
    COMMAND ${CMAKE_COMMAND}
      -DHOST=$<TARGET_FILE:win10xbox>
      -DMACHINE=${_machine}
      -DNAME=${name}
      -DEXPECT=${expect}
      -DMAX_FAILURES=${max_failures}
      -DTIMEOUT=${WIN10XBOX_WINE_TEST_TIMEOUT}
      -DRESULTS_DIR=${_results_dir}
      -DEMULATOR=${CMAKE_CROSSCOMPILING_EMULATOR}
      -P ${CMAKE_CURRENT_SOURCE_DIR}/run_wine_test.cmake)
  math(EXPR _ctest_timeout "${WIN10XBOX_WINE_TEST_TIMEOUT} + 30")
  set_tests_properties(wine_conformance.${dll}.${test} PROPERTIES
    FIXTURES_REQUIRED "w10x_machine;wine_staged"
    SKIP_RETURN_CODE 77
    SKIP_REGULAR_EXPRESSION "SKIP:"
    LABELS "wine_conformance"
    TIMEOUT ${_ctest_timeout})
endfunction()

# ---- staging + report --------------------------------------------------------
get_property(_guests GLOBAL PROPERTY WIN10XBOX_WINE_GUESTS)
set(_stage_args
  -DMACHINE=${_machine}
  -DGUEST_DIR=${_guest_dir}
  -DRESULTS_DIR=${_results_dir}
  -P ${CMAKE_CURRENT_SOURCE_DIR}/stage_wine_tests.cmake)
add_custom_target(wine_tests COMMENT "Wine conformance guests built")
if(_guests)
  add_dependencies(wine_tests ${_guests})
endif()
add_custom_target(wine_tests_stage COMMAND ${CMAKE_COMMAND} ${_stage_args}
  COMMENT "Staging Wine conformance guests into ${_machine}/guest-wine" VERBATIM)
add_dependencies(wine_tests_stage wine_tests)
add_custom_target(wine_tests_report
  COMMAND ${CMAKE_COMMAND} -DRESULTS_DIR=${_results_dir} -P ${CMAKE_CURRENT_SOURCE_DIR}/report.cmake
  COMMENT "Wine conformance results" VERBATIM)

add_test(NAME wine_conformance.stage COMMAND ${CMAKE_COMMAND} ${_stage_args})
set_tests_properties(wine_conformance.stage PROPERTIES
  FIXTURES_REQUIRED w10x_machine FIXTURES_SETUP wine_staged)
# Runs after all wine tests (fixture cleanup): prints the result table.
add_test(NAME wine_conformance.report
  COMMAND ${CMAKE_COMMAND} -DRESULTS_DIR=${_results_dir} -P ${CMAKE_CURRENT_SOURCE_DIR}/report.cmake)
set_tests_properties(wine_conformance.report PROPERTIES
  FIXTURES_CLEANUP wine_staged LABELS "wine_conformance")
set_tests_properties(wine_conformance.stage PROPERTIES LABELS "wine_conformance")
